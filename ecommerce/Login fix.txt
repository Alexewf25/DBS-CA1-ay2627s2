Step 3: Fix the Login Debug Issue
The login might also be silently failing due to missing error handling in the front end. Replace your public/auth/login/script.js with this improved version that shows errors clearly:

// public/auth/login/script.js

document.getElementById("loginForm").addEventListener("submit", async (e) => {
  e.preventDefault();

  const username = document.getElementById("username").value.trim();
  const password = document.getElementById("password").value;
  const msgEl = document.getElementById("message");
  const submitBtn = e.target.querySelector("button[type='submit']");

  if (!username || !password) {
    showMessage(msgEl, "Please enter both username and password.", "error");
    return;
  }

  // Disable button to prevent double submit
  submitBtn.disabled = true;
  submitBtn.textContent = "Logging in...";

  try {
    const res = await fetch("/api/auth/login", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ username, password }),
    });

    // Check if response is ok
    if (!res.ok && res.status !== 401) {
      showMessage(msgEl, `Server error: ${res.status} ${res.statusText}`, "error");
      submitBtn.disabled = false;
      submitBtn.textContent = "Login";
      return;
    }

    const data = await res.json();

    if (data.success) {
      localStorage.setItem("token", data.token);
      localStorage.setItem("memberId", data.member.id);
      localStorage.setItem("username", data.member.username);
      localStorage.setItem("role", data.member.role);

      showMessage(msgEl, "Login successful! Redirecting...", "success");

      setTimeout(() => {
        if (parseInt(data.member.role) === 1) {
          window.location.href = "/admin/dashboard/salesOrderSummary";
        } else {
          window.location.href = "/product/retrieve";
        }
      }, 1000);

    } else {
      showMessage(msgEl, data.message || "Login failed. Please try again.", "error");
      submitBtn.disabled = false;
      submitBtn.textContent = "Login";
    }

  } catch (err) {
    console.error("Login error:", err);
    showMessage(msgEl, "Cannot connect to server. Is the app running?", "error");
    submitBtn.disabled = false;
    submitBtn.textContent = "Login";
  }
});

function showMessage(el, msg, type) {
  el.textContent = msg;
  el.className = `message ${type}`;
}

function copyText(value, fieldId) {
  document.getElementById(fieldId).value = value;
  navigator.clipboard.writeText(value).catch(() => {
    const el = document.createElement("textarea");
    el.value = value;
    document.body.appendChild(el);
    el.select();
    document.execCommand("copy");
    document.body.removeChild(el);
  });
}

Step 5: Add Console Logging to authController to Debug
Temporarily update controllers/authController.js to log what is happening:

const bcrypt = require("bcrypt");
const jwt = require("jsonwebtoken");
const authModel = require("../models/authModel");

const login = async (req, res) => {
  const { username, password } = req.body;

  console.log("=== LOGIN ATTEMPT ===");
  console.log("Username received:", username);
  console.log("Password received:", password ? "YES (hidden)" : "NO");

  try {
    if (!username || !password) {
      return res.status(400).json({
        success: false,
        message: "Username and password are required.",
      });
    }

    const member = await authModel.findByUsername(username);
    console.log("Member found in DB:", member ? "YES" : "NO");

    if (!member) {
      return res.status(401).json({
        success: false,
        message: "Invalid username or password.",
      });
    }

    console.log("Stored hash:", member.password);

    const isMatch = await bcrypt.compare(password, member.password);
    console.log("Password match:", isMatch);

    if (!isMatch) {
      return res.status(401).json({
        success: false,
        message: "Invalid username or password.",
      });
    }

    const token = jwt.sign(
      {
        id: member.id,
        username: member.username,
        role: member.role,
      },
      process.env.JWT_SECRET,
      { expiresIn: process.env.JWT_EXPIRES_IN }
    );

    console.log("Login successful for:", member.username);
    console.log("Role:", member.role);

    res.json({
      success: true,
      token,
      member: {
        id: member.id,
        username: member.username,
        role: member.role,
      },
    });

  } catch (err) {
    console.error("LOGIN ERROR:", err);
    res.status(500).json({
      success: false,
      message: "Server error: " + err.message,
    });
  }
};

const register = async (req, res) => {
  const { username, email, dob, password, gender } = req.body;
  try {
    if (!username || !email || !dob || !password || !gender) {
      return res.status(400).json({
        success: false,
        message: "All fields are required.",
      });
    }
    const existing = await authModel.findByUsername(username);
    if (existing) {
      return res.status(409).json({
        success: false,
        message: "Username already exists.",
      });
    }
    const hashedPassword = await bcrypt.hash(password, 10);
    const member = await authModel.createMember(
      username, email, dob, hashedPassword, gender
    );
    res.status(201).json({
      success: true,
      message: "Registration successful.",
      member,
    });
  } catch (err) {
    console.error("REGISTER ERROR:", err);
    res.status(500).json({
      success: false,
      message: "Server error: " + err.message,
    });
  }
};

module.exports = { login, register };
